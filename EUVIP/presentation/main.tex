\begin{frame}[t]{The Hessian describes local ridge geometry}
\vspace{-1mm}
\[
 \Hess_\sigma(x)=\nabla^2(G_\sigma*I)(x),\qquad |\lambda_1|\leq|\lambda_2|.
\]
\vspace{-2mm}
\begin{columns}[c,onlytextwidth]
\begin{column}{.43\textwidth}\centering
\resizebox{.86\linewidth}{!}{%
\begin{tikzpicture}[x=1cm,y=1cm,

 \begin{scope}[rotate=27]
  \draw[ridge](-2.05,0)--(2.05,0);
  \draw[inria-rouge,line width=1pt,fill=inria-rouge!8](0,0)ellipse(1.75 and .62);
  \draw[vec,inria-rouge](0,0)--(1.833,0);
  \draw[vec,inria-rouge!62](0,0)--(0,.703);
  \node[font=\small,inria-rouge,fill=white,inner sep=1pt]at(.95,0){$e_1$};
  \node[font=\small,inria-rouge!62,fill=white,inner sep=1pt]at(0,.4){$e_2$};
 \end{scope}
 \draw[gray!65,dashed](0,0)--(2.2,0);
 \draw[gris_fonce_inria!80](1.98,0)arc(0:27:1.98);
 \node[font=\small]at(1.82,.45){$\theta$};
 \fill[inria-2024-bleu-canard](0,0)circle(2.1pt);
 \node[below left,font=\small]at(0,0){$x$};
 \draw[inria-rouge](1.559,.795)--(1.66,1.04);
 \node[font=\small,inria-rouge,anchor=south]at(1.66,1.03){$\dfrac1{\sqrt{|\lambda_1|}}$};
 \draw[inria-rouge!62](-.282,.552)--(-.62,.9);
 \node[font=\small,inria-rouge!62,anchor=south east]at(-.58,.88){$\dfrac1{\sqrt{|\lambda_2|}}$};
\end{tikzpicture}}
\end{column}
\begin{column}{.53\textwidth}
 \textbf{Shape}\quad $R_B=|\lambda_1|/|\lambda_2|$\par
 {\small Near 0: elongated ridge. Near 1: blob.}\par\vspace{3mm}
 \textbf{Contrast}\quad $S=\sqrt{\lambda_1^2+\lambda_2^2}$\par
 {\small Strength of the local curvature.}\par\vspace{3mm}
 \textbf{Direction}\quad $e_1$ runs along the ridge.
\end{column}
\end{columns}
\vspace{2mm}
\begin{beamercolorbox}[sep=1.5mm,center]{normal text}
$F_\sigma(x)=\exp\!\left(-\dfrac{R_B^2}{2\beta^2}\right)
 \left(1-\exp\!\left(-\dfrac{S^2}{2c^2}\right)\right)$ if $\lambda_2>0$; otherwise $0$.
\par\vspace{1mm}{\small Classical \Frangi{}: one score per pixel; retain the best scale.}
\end{beamercolorbox}
\source{\cit{Frangi 98} A. F. Frangi et al., ``Multiscale vessel enhancement filtering,'' MICCAI, 1998. Diagram adapted from the thesis defense.}
\end{frame}

\begin{frame}[t]{What the graph adds: orientation alignment}
\vspace{1mm}
The filter scores pixels separately; the graph checks \textbf{compatibility between pixels}.
\par\vspace{2mm}\centering
\resizebox{.71\linewidth}{!}{%
\begin{tikzpicture}[x=1cm,y=1cm,
 ridge/.style={line width=7pt,gris_fonce_inria!45,line cap=round},
 dir/.style={-{Latex[length=1.9mm]},line width=1pt},
 pix/.style={circle,fill=inria-2024-bleu-canard,inner sep=0pt,minimum size=3.8pt}]
 \draw[ridge](-.15,-.3)--(2.75,.62);
 \node[pix](i1)at(.72,-.02){};\node[pix](j1)at(1.88,.35){};
 \draw[dir,inria-rouge](i1)--++(.62,.2);\draw[dir,inria-rouge](j1)--++(.62,.2);
 \draw[inria-rouge,line width=1.6pt](i1)--(j1);
 \node[font=\scriptsize]at(.6,-.36){$\theta(x_i)$};\node[font=\scriptsize]at(2.16,.02){$\theta(x_j)$};
 \node[font=\footnotesize,align=center]at(1.3,1.15){Aligned axes\\\textbf{strong link}};
 \begin{scope}[shift={(4.35,0)}]
  \draw[ridge](-.15,-.3)--(2,.4);\draw[ridge](1.3,.92)--(2.32,-.48);
  \node[pix](i2)at(.72,-.02){};\node[pix](j2)at(1.66,.43){};
  \draw[dir,inria-rouge](i2)--++(.59,.19);\draw[dir,inria-rouge](j2)--++(.36,-.5);
  \draw[gris_fonce_inria!45,line width=.8pt](i2)--(j2);
  \node[font=\scriptsize]at(.6,-.36){$\theta(x_i)$};\node[font=\scriptsize,anchor=west]at(2.1,-.14){$\theta(x_j)$};
  \node[font=\footnotesize,align=center]at(1.2,1.15){Incompatible axes\\\textbf{weak link}};
 \end{scope}
\end{tikzpicture}}
\par\vspace{2mm}
$\displaystyle S^\sigma_{\mathrm{align}}(i,j)=\exp\!\left[-\frac12
 \left(\frac{\sin(\theta^\sigma(x_i)-\theta^\sigma(x_j))}{s_{\mathrm a}}\right)^2\right]$
\par\vspace{2mm}{\small The sine respects unoriented axes: $\theta$ and $\theta+\pi$ are equivalent.}
\par\vspace{2mm}
$\displaystyle S^{(0)}_{ij}=\max_{\sigma\in\Sigma}
 \underbrace{S^\sigma_{\mathrm{int}}S^\sigma_{\mathrm{shape}}}_{\text{curvature and elongation}}
 \underbrace{S^\sigma_{\mathrm{align}}}_{\text{pairwise refinement}}$
\source{\owncit{GRETSI 25; EUVIP 26} L. Hauseux et al. Curvature and elongation are refined by orientation compatibility. Diagram adapted from the defense.}
\end{frame}

% 4. Operator-level fusion.
\begin{frame}[t]{Fuse modalities before measuring ridge geometry}
\vspace{1mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.55\textwidth}
 \textbf{1. Normalize each Hessian}\par\vspace{-2mm}
 \[
 \widehat\Hess_\sigma^{(m)}(x)=
 \frac{\Hess_\sigma^{(m)}(x)}{\max_{y\in\Omega}\|\Hess_\sigma^{(m)}(y)\|_2}.
 \]
 \textbf{2. Fuse the normalized operators}\par\vspace{-2mm}
 \[
 \Hess_\sigma^{\mathrm{fused}}(x)=\sum_m w_m\widehat\Hess_\sigma^{(m)}(x).
 \]
 {\small $w_m\geq0$, $\sum_m w_m=1$. Normalization is per modality and per scale.}
\end{column}
\begin{column}{.40\textwidth}
 \panel{.475}{PalaisDesPapes.png}{Intensity}\hfill
 \panel{.475}{PalaisDesPapes_IntDepth.png}{Depth overlay}
 \par\vspace{3mm}
 {\small Aligned modalities share the crack geometry; sensor-specific artifacts may differ.}
 \par\vspace{2mm}{\scriptsize Palais des Papes, France. \textcopyright\ Cerema.}
\end{column}
\end{columns}
\vspace{2mm}
\takeaway{Compute \textbf{eigenvalues and directions after fusion} --- do not average \Frangi{} response maps.}
\source{\owncit{EUVIP 26} Section III-B. $\|\cdot\|_2$: spectral norm. Aligned modalities and consistent ridge polarity are required.}
\end{frame}
