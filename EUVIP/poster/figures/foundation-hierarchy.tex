% 3. HSA block constraint (Amizadeh et al., NeurIPS 2025, section 3.2).
% One attention weight is shared between all leaves of two sibling subtrees.
% This is an attention matrix, not an additive bias or a learned diagonal prior.
% Row/column order: a1,a2,b1,b2,c1,c2. The diagonal is zero.
% Illustrative coefficients, not measurements, chosen to sum to 1 per row:
% A-internal=.14, A->B=.25, B->A=.10, B-internal=.44,
% AB->C=.18, C->AB=.22, C-internal=.12 (values are per matrix entry).
% The example has no systematic preference for interactions inside a group.
% Distinct directions may have distinct weights; AB->C is one 4x2 block.
\begin{scope}[shift={(10.9,1.15)},x=.45cm,y=.45cm]
  % Highest sibling interaction: subtree AB with subtree C.
  \fill[frangiOutlookBlue,fill opacity=.4091] (4,2) rectangle (6,6);
  \fill[frangiOutlookBlue,fill opacity=.50] (0,0) rectangle (4,2);
  % Sibling interaction inside subtree AB.
  \fill[frangiOutlookBlue,fill opacity=.5682] (2,4) rectangle (4,6);
  \fill[frangiOutlookBlue,fill opacity=.2273] (0,2) rectangle (2,4);
  % Two leaves per branch: only the off-diagonal entries receive a weight.
  \foreach \x/\y in {0/4,1/5}
    \fill[frangiOutlookBlue,fill opacity=.3182] (\x,\y) rectangle ++(1,1);
  \foreach \x/\y in {2/2,3/3}
    \fill[frangiOutlookBlue,fill opacity=1] (\x,\y) rectangle ++(1,1);
  \foreach \x/\y in {4/0,5/1}
    \fill[frangiOutlookBlue,fill opacity=.2727] (\x,\y) rectangle ++(1,1);
  % A subtle grid exposes six tokens without white cell borders.
  \draw[step=1,line width=.20pt,frangiOutlookNavy!22] (0,0) grid (6,6);
  \draw[line width=.45pt,frangiOutlookNavy!60] (0,0) rectangle (6,6);

  \draw[line width=.65pt,frangiOutlookNavy!60] (0,2)--(6,2);
  \draw[line width=.65pt,frangiOutlookNavy!60] (2,2)--(2,6);
  \draw[line width=.65pt,frangiOutlookNavy!60] (0,4)--(4,4);
