\begin{frame}{La tolérance de 3 px : en quoi elle consiste}
  \centering
  \vspace{0.04cm}
  {\small $\mathrm{Pr}_k = \dfrac{|P \cap G^{\oplus k}|}{|P|}$ \quad
   $\mathrm{Re}_k = \dfrac{|G \cap P^{\oplus k}|}{|G|}$ \quad
   $\IoU_k = \dfrac{F_k}{2 - F_k}$, \;
   $F_k = \dfrac{2\,\mathrm{Pr}_k \mathrm{Re}_k}{\mathrm{Pr}_k + \mathrm{Re}_k}$}
