        \draw[->,thick] (memory) -- (decoder);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{Ajouts techniques} \refmark{2}
      \begin{itemize}
        \item mémoire continue pour suivre un objet ;
        \item connexions haute résolution vers le décodeur ;
        \item tête indiquant si l'objet est présent ;
        \item entraînement commun sur images et vidéos.
      \end{itemize}

    \end{column}
  \end{columns}

  \centering
  \metriccard{inkblue}{Segmentation d'image}{plus précis}{et $6\times$ plus rapide que SAM}
  \hspace{0.35cm}
  \metriccard{success}{Segmentation vidéo}{$3\times$ moins}{d'interactions que les méthodes comparées}
  \source{\refmark{2} Ravi et al., \emph{SAM 2: Segment Anything in Images and Videos}, 2024.}

\begin{frame}{Pourquoi nous utilisons SAM 2}
  \centering
  \vspace{0.1cm}
  {\large Notre tâche est une segmentation automatique, binaire, sur des images fixes.}\par
  \vspace{0.55cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.30\textwidth}
      \centering
      {\Large\bfseries\color{inkblue} Tâche}\\[5pt]
      {\small SAM 2 s'adapte directement à une classe connue. SAM 3 vise toutes les instances d'un concept.}
    \end{column}
    \begin{column}{0.30\textwidth}
      \centering
      {\Large\bfseries\color{warning} Représentations}\\[5pt]
      {\small Les représentations Hiera-L sont déjà utilisées par notre baseline et par le correcteur Frangi.}
    \end{column}
    \begin{column}{0.30\textwidth}
      \centering
      {\Large\bfseries\color{success} Comparaison}\\[5pt]
      {\small Passer à SAM 3 changerait simultanément l'encodeur, la tâche et l'interface d'entrée.}
    \end{column}
  \end{columns}

  \vspace{0.65cm}
  \begin{block}{Choix}
    \centering\large Nous poursuivons l'intégration de Frangi avec \textbf{SAM 2}. SAM 3 reste une étude séparée.
  \end{block}
  \source{\refmark{2} Ravi et al., 2024. \quad
  \refmark{3} Carion et al., ICLR 2026.}
\end{frame}
