\begin{frame}{Frangi--similarité : définition exacte du score}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.65\textwidth}
      \small
      Pour une arête $(i,j)$ et une échelle $\sigma$, on pose
      {\scriptsize $\lambda_1^\sigma,\lambda_2^\sigma,\theta^\sigma$ issus du Hessien de luminance normalisé par sa norme spectrale maximale à cette échelle.}
      \[
        r_i^\sigma=\frac{|\lambda_1^\sigma(i)|}{|\lambda_2^\sigma(i)|+\varepsilon},
        \qquad
        c_i^\sigma=\lambda_2^\sigma(i)\,\mathbf{1}_{\lambda_2^\sigma(i)>0}.
      \]
      \vspace{-0.12cm}
      \[
      \begin{aligned}
        S_{ij}^{\sigma}
        ={}&
        \underbrace{\exp\!\left[-\frac12
          \left(\frac{r_i^\sigma+r_j^\sigma}{s_s}\right)^2\right]}_{\text{forme allongée}}\\[-1pt]
        &\times
        \underbrace{\left(1-\exp\!\left[-\frac12
          \left(\frac{c_i^\sigma c_j^\sigma}{s_i}\right)^2\right]\right)}_{\text{force aux deux extrémités}}\\[-1pt]
        &\times
        \underbrace{\exp\!\left[-\frac12
          \left(\frac{\sin(\theta_i^\sigma-\theta_j^\sigma)}{s_a}\right)^2\right]}_{\text{alignement non orienté}} .
      \end{aligned}
      \]
      \vspace{-0.12cm}
      \[
        S_{ij}=\max_{\sigma\in\Sigma}S_{ij}^{\sigma},
        \qquad
        P_i=\max_{j:(i,j)\in E_{\mathrm{retenu}}}S_{ij}.
      \]
    \end{column}
    \begin{column}{0.32\textwidth}
      \centering
      \includegraphics[width=0.90\linewidth,height=0.40\textheight,keepaspectratio]{find_similarity.png}

      \vspace{0.12cm}
      \begin{block}{Paramètres de l'essai}
        \fontsize{6.5}{7.5}\selectfont
        entrée : luminance visible normalisée\\
        $\Sigma=\{1,3,5,9,15\}$ px,\quad $R=3$ px\\
        $s_s=1,\ s_i=0{,}25,\ s_a=0{,}3$\\
        $18\,\%$ des arêtes et n\oe{}uds les plus forts conservés
      \end{block}
      \vspace{0.06cm}
      \raggedright\scriptsize
      $P_i\in[0,1]$ est une \textbf{évidence géométrique}. Ce n'est pas une probabilité de fissure apprise.
    \end{column}
  \end{columns}
  \source{Implémentation \texttt{extract\_frangi\_graph\_gpu} utilisée pour les prompts CrackSAM 2.}
\end{frame}

      \vspace{0.16cm}

    \end{column}
    \begin{column}{0.66\textwidth}
      \centering
      \vspace{0.16cm}
      \begin{tikzpicture}[>=Latex,node distance=0.22cm]
        \node[rounded corners=5pt,draw=inkblue!50,fill=softblue,

        \node[rounded corners=5pt,draw=warning!55,fill=warning!8,

          {\scriptsize $\widetilde P=\operatorname{clip}(P,\varepsilon,1-\varepsilon)$}\\[-1pt]
          $\displaystyle L=\log\frac{\widetilde P}{1-\widetilde P}$\\[-1pt]
          {\tiny $\varepsilon=10^{-5}$}};
        \node[rounded corners=5pt,draw=black!35,fill=softgray,
          minimum width=1.55cm,minimum height=1.12cm,align=center,right=of logit] (resize) {
          {\bfseries $256\times256$}\\{\scriptsize interpolation}};
        \node[rounded corners=5pt,draw=failure!55,fill=softred,
          minimum width=1.75cm,minimum height=1.12cm,align=center,right=of resize] (mask) {
          {\bfseries \texttt{mask\_input}}\\{\scriptsize SAM 2}};
        \draw[->,very thick] (p) -- (logit);
        \draw[->,very thick] (logit) -- (resize);
        \draw[->,very thick] (resize) -- (mask);
      \end{tikzpicture}

      \vspace{0.48cm}
      \begin{columns}[T,onlytextwidth]
        \begin{column}{0.47\textwidth}
          \begin{block}{Présence}
            \small Un grand $P$ indique une structure localement compatible avec une fissure.
          \end{block}
        \end{column}
        \begin{column}{0.47\textwidth}
          \begin{alertblock}{Absence}
            \small $P\simeq0$ devient $L\simeq-11{,}5$ : une forte indication de fond, jamais neutre.
          \end{alertblock}
        \end{column}
      \end{columns}

      \vspace{0.12cm}
      \small Le même prompt dense est présent pendant tout l'entraînement et à chaque inférence.
    \end{column}
  \end{columns}
  \source{Protocole historique Frangi--similarité, meilleur checkpoint à l'époque 25.}
\end{frame}
